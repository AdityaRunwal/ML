Step 2: Import Libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.datasets import fetch_california_housing
from sklearn.model_selection import train_test_split

from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score



Step 3: Load the Dataset
housing = fetch_california_housing(as_frame=True)
df = housing.frame

Step 6: Separate Features and Target

Step 7: Train-Test Split
X_train, X_test, y_train, y_test = train_test_split( X, y, test_size=0.20, random_state=42)

print("Testing data:", X_test.shape)



Step 8: Feature Scaling
scaler = StandardScaler()

Step 11: Train Linear Regression
linear_model = LinearRegression()

Step 20: Effect of Different Alpha Values
alphas = [0.001, 0.01, 0.1, 1, 10, 100]

ridge_scores = []
lasso_scores = []

for alpha in alphas:

    ridge = Ridge(alpha=alpha)
    ridge.fit(X_train_scaled, y_train)
    ridge_pred = ridge.predict(X_test_scaled)
    ridge_scores.append( r2_score(y_test, ridge_pred) )

    lasso = Lasso(alpha=alpha, max_iter=10000)
    lasso.fit(X_train_scaled, y_train)
    lasso_pred = lasso.predict(X_test_scaled)
    lasso_scores.append(r2_score(y_test, lasso_pred))



Step 21: Plot Effect of Regularization Strength
plt.figure(figsize=(10, 6))

plt.semilogx( alphas,ridge_scores, marker="o",label="Ridge")
plt.semilogx(alphas, lasso_scores, marker="s",label="Lasso")

plt.xlabel("Alpha")
plt.ylabel("Test R² Score")
plt.title("Effect of Regularization Strength")
plt.legend()
plt.grid(True)
plt.show()
