Step 1: Import Required Libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error, r2_score


Step 2: Load the Dataset
data = {
    "YearsExperience": [1.2, 1.4, 1.6, 2.1, 2.3, 3.0, 3.1, 3.3, 3.3, 3.8],
    
    "Salary": [39344.0, 46206.0, 37732.0, 43526.0, 39892.0,
               56642.0, 60151.0, 54446.0, 64446.0, 57191.0]
}

df = pd.DataFrame(data)

print(df)


Step 3: Define Input and Output
X = df[["YearsExperience"]]
y = df["Salary"]


Step 4: Split the Dataset
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.3,
    random_state=42
)

print("X_train shape:", X_train.shape)
print("X_test shape:", X_test.shape)


Step 5: Train Polynomial Regression Models
degrees = [1, 2, 3, 4, 5]

results = []

plt.figure(figsize=(18, 10))

for i, degree in enumerate(degrees):

    # Create polynomial features
    poly = PolynomialFeatures(degree=degree)

    X_train_poly = poly.fit_transform(X_train)
    X_test_poly = poly.transform(X_test)

    # Train model
    model = LinearRegression()
    model.fit(X_train_poly, y_train)

    # Predictions
    y_train_pred = model.predict(X_train_poly)
    y_test_pred = model.predict(X_test_poly)

    # Calculate RMSE
    train_rmse = np.sqrt(mean_squared_error(y_train, y_train_pred))
    test_rmse = np.sqrt(mean_squared_error(y_test, y_test_pred))

    # Calculate R2 score
    train_r2 = r2_score(y_train, y_train_pred)
    test_r2 = r2_score(y_test, y_test_pred)

    # Save results
    results.append([
        degree,
        train_rmse,
        test_rmse,
        train_r2,
        test_r2
    ])

    # Create subplot
    plt.subplot(2, 3, i + 1)

    # Plot actual points
    plt.scatter(X_train, y_train, label="Training Data")

    # Create smooth curve
    X_plot = np.linspace(X.min().iloc[0], X.max().iloc[0], 100).reshape(-1, 1)

    X_plot_poly = poly.transform(X_plot)
    y_plot = model.predict(X_plot_poly)

    plt.plot(X_plot, y_plot)

    plt.title(f"Degree {degree}")
    plt.xlabel("Years of Experience")
    plt.ylabel("Salary")
    plt.legend()

plt.tight_layout()
plt.show()


Step 6: Display Results Table
results_df = pd.DataFrame(
    results,
    columns=[
        "Degree",
        "Train RMSE",
        "Test RMSE",
        "Train R2",
        "Test R2"
    ]
)

print(results_df)
