Step 1: Import Libraries
import numpy as np
import matplotlib.pyplot as plt
import random

print("Libraries imported successfully!")



Step 2: Create the Grid World Environment
ROWS = 5
COLS = 5

START = (0, 0)
GOAL = (4, 4)

OBSTACLES = [(1, 1), (2, 1), (3, 3)]

ACTIONS = {
    0: (-1, 0),   # Up
    1: (1, 0),    # Down
    2: (0, -1),   # Left
    3: (0, 1)     # Right
}

ACTION_NAMES = {
    0: "Up",
    1: "Down",
    2: "Left",
    3: "Right"
}

print("Grid World environment created!")
print("Start:", START)
print("Goal:", GOAL)
print("Obstacles:", OBSTACLES)


Step 3: Define the Environment Step Function
def step(state, action):
    row, col = state
    dr, dc = ACTIONS[action]

    new_row = row + dr
    new_col = col + dc

    # Check if the agent goes outside the grid
    if not (0 <= new_row < ROWS and 0 <= new_col < COLS):
        return state, -5, False

    next_state = (new_row, new_col)

    # Check if the agent hits an obstacle
    if next_state in OBSTACLES:
        return state, -10, False

    # Reward for reaching the goal
    if next_state == GOAL:
        return next_state, 100, True

    # Small penalty for each ordinary move
    return next_state, -1, False


print("Environment step function defined successfully!")



Step 4: Initialize the Q-Table and Parameters
Q = np.zeros((ROWS, COLS, len(ACTIONS)))

alpha = 0.1
gamma = 0.95
epsilon = 1.0
epsilon_min = 0.01
epsilon_decay = 0.995

episodes = 3000
max_steps = 100

print("Q-table shape:", Q.shape)
print("Episodes:", episodes)
print("Learning rate:", alpha)
print("Discount factor:", gamma)


Step 5: Train the Q-Learning Agent
episode_rewards = []
success_history = []
epsilon_history = []

for episode in range(episodes):

    state = START
    total_reward = 0
    reached_goal = False

    for _ in range(max_steps):

        # Epsilon-greedy action selection
        if random.random() < epsilon:
            action = random.choice(list(ACTIONS.keys()))
        else:
            action = int(np.argmax(Q[state[0], state[1]]))

        # Take action
        next_state, reward, done = step(state, action)

        # Get old Q-value
        old_value = Q[state[0], state[1], action]

        # Get maximum Q-value for next state
        next_max = np.max(Q[next_state[0], next_state[1]])

        # Q-Learning formula
        new_value = old_value + alpha * (
            reward + gamma * next_max - old_value
        )

        # Update Q-table
        Q[state[0], state[1], action] = new_value

        # Move to next state
        state = next_state
        total_reward += reward

        if done:
            reached_goal = True
            break

    # Store results
    episode_rewards.append(total_reward)
    success_history.append(int(reached_goal))

    # Reduce exploration rate
    epsilon = max(epsilon_min, epsilon * epsilon_decay)
    epsilon_history.append(epsilon)


print("Q-Learning training completed!")
print("Final epsilon:", round(epsilon, 4))
print("Goal reached in the final episode:", bool(success_history[-1]))


Step 6: Plot Average Reward Per Episode
window = 100

average_rewards = np.convolve(
    episode_rewards,
    np.ones(window) / window,
    mode="valid"
)

plt.figure(figsize=(10, 5))

plt.plot(average_rewards)

plt.title("Q-Learning: Average Reward per Episode")
plt.xlabel("Episode")
plt.ylabel("Average Reward (100-episode window)")
plt.grid(True)

plt.show()


Step 7: Create the Learned Policy
ARROWS = {
    0: "↑",
    1: "↓",
    2: "←",
    3: "→"
}

policy = np.full((ROWS, COLS), "", dtype="<U2")

for row in range(ROWS):
    for col in range(COLS):

        state = (row, col)

        if state == START:
            policy[row, col] = "S"

        elif state == GOAL:
            policy[row, col] = "G"

        elif state in OBSTACLES:
            policy[row, col] = "X"

        else:
            best_action = int(np.argmax(Q[row, col]))
            policy[row, col] = ARROWS[best_action]


print("Learned Optimal Policy:")
print("S = Start, G = Goal, X = Obstacle")

for row in policy:
    print(" ".join(row))


Step 8: Visualize the Learned Q-Learning Policy
grid = np.zeros((ROWS, COLS))

# Mark obstacles
for r, c in OBSTACLES:
    grid[r, c] = -1

# Mark goal
grid[GOAL] = 1

fig, ax = plt.subplots(figsize=(7, 7))

ax.imshow(grid, cmap="coolwarm", vmin=-1, vmax=1)

for row in range(ROWS):
    for col in range(COLS):
        ax.text(
            col,
            row,
            policy[row, col],
            ha="center",
            va="center",
            fontsize=20,
            color="black"
        )

ax.set_xticks(range(COLS))
ax.set_yticks(range(ROWS))

ax.set_xticks(np.arange(-0.5, COLS, 1), minor=True)
ax.set_yticks(np.arange(-0.5, ROWS, 1), minor=True)

ax.grid(which="minor", color="black", linewidth=1.5)

ax.tick_params(which="minor", bottom=False, left=False)

plt.title("Learned Q-Learning Policy\nS: Start | G: Goal | X: Obstacle")

plt.show()


Step 9: Test the Learned Path
state = START
path = [state]
total_reward = 0

for _ in range(ROWS * COLS * 2):

    if state == GOAL:
        break

    action = int(np.argmax(Q[state[0], state[1]]))

    next_state, reward, done = step(state, action)

    # Stop if agent is stuck
    if next_state == state:
        print("Agent is stuck at:", state)
        break

    state = next_state
    path.append(state)
    total_reward += reward

    if done:
        break


print("\nLearned Path:")
print(path)

print("Total Reward:", total_reward)
print("Total Moves:", len(path) - 1)
print("Goal reached:", state == GOAL)


Step 10: Display Final Training Results
last_100_successes = success_history[-100:]
success_rate = np.mean(last_100_successes) * 100

print("\n========== FINAL TRAINING RESULTS ==========")

print("Total training episodes:", episodes)

print("Success rate in last 100 episodes:",
